Campaign paginate List
=============================
query GetCampaignList($page: Int!, $perPage: Int!) {
  campaignList(page: $page, perPage: $perPage) {
    data {
      node {
        name
        title
      }
    }
    pageInfo {
      total
      perPage
      currentPage
      lastPage
    }
  }
}

{
  "page": 1,
  "perPage": 2
}

Campaign Update by ID
==========================
mutation UpdateCampaign($_id: ID!, $input: UpdateCampaignInput!) {
  updateCampaign(_id: $_id, input: $input) {
    success
    message
    campaign {
      _id
      name
      title
      description
      effective_dates {
        startDate
        endDate
      }
      delivery_method {
        email
        phone
      }
      stores
      tags
    }
  }
}
